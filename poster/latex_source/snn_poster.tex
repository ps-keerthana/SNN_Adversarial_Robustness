\begin{block}{Introduction}
  \colorbox{boxbg}{\parbox{\dimexpr\linewidth-10pt\relax}{%
    \small\bfseries Does spike encoding strategy affect adversarial
    robustness in SNNs --- and do SNNs resist attacks differently from ANNs?}}

  \vspace{0.4em}
  \textbf{\color{accentblue}Why This Matters}
  \begin{itemize}
    \item SNNs increasingly deployed on edge \& neuromorphic hardware for safety-critical tasks
    \item Adversarial robustness of SNNs remains \emph{understudied} vs.\ classical deep nets
    \item Encoding strategy (rate vs.\ temporal) may introduce implicit robustness --- but no systematic study exists
  \end{itemize}
\end{block}

\begin{block}{Encoding Strategies}
  \begin{tikzpicture}
    \node[fill=snnrate!18,rounded corners=3pt,inner sep=6pt,
          text width=0.43\linewidth,align=left] (r){%
      \textbf{\color{snnrate}Rate Coding}\\[0.12em]
      \footnotesize Pixel intensity $\to$ spike \textbf{frequency} over $T$ steps.
      Bernoulli sampling. Dense.};
    \node[fill=snntemporal!18,rounded corners=3pt,inner sep=6pt,
          text width=0.43\linewidth,align=left,right=0.5em of r]{%
      \textbf{\color{snntemporal}Temporal Coding}\\[0.12em]
      \footnotesize Intensity $\to$ \textbf{first-spike time}.
      Bright pixels spike early. Sparse, timing-dependent.};
  \end{tikzpicture}

  \vspace{0.4em}
  {\footnotesize\begin{tabular}{@{}ll@{}}
    \toprule
    \textbf{Property}        & \textbf{Robustness implication} \\
    \midrule
    Rate: Bernoulli noise    & Natural stochastic regulariser \\
    Temporal: single spike   & Gradient harder to exploit \\
    Surrogate gradient       & Enables backprop through spikes \\
    \bottomrule
  \end{tabular}}
\end{block}

\begin{block}{Architecture}
  \textbf{\color{accentblue}SNN-MNIST (LIF-MLP)}\\[0.15em]
  {\small Input(784) $\to$ FC(1000) $\to$ LIF $\to$ FC(10) $\to$ LIF\quad
  $T\!=\!25$, $\beta\!=\!0.9$, Adam $2\!\times\!10^{-3}$}

  \vspace{0.4em}
  \textbf{\color{accentblue}SpikingCIFAR (Conv-SNN)}\\[0.15em]
  {\small Conv(64)$\to$BN$\to$LIF $\to$ Conv(128)$\to$BN$\to$LIF $\to$
  Conv(256)$\to$BN$\to$LIF $\to$ Dropout(0.4) $\to$ FC(512)$\to$BN$\to$LIF
  $\to$ FC(10)$\to$LIF\\[0.1em]
  $T_\text{rate}\!=\!10$,\ $T_\text{temp}\!=\!25$ $\cdot$ Cosine LR $\cdot$ 60 epochs}

  \vspace{0.4em}
  \textbf{\color{accentblue}Training}\\[0.15em]
  {\small Fast-sigmoid surrogate gradient (Neftci et al.\ 2019) via snnTorch.
  Adam optimiser, cosine LR. 5 runs (MNIST), 1 run (CIFAR-10).}
\end{block}

\begin{block}{Adversarial Attacks}
  \begin{center}
  \begin{tikzpicture}[
    box/.style={draw=accentblue,fill=lightgray,rounded corners=3pt,
                minimum height=0.95cm,inner sep=4pt,
                font=\footnotesize\bfseries,text width=1.8cm,align=center},
    arr/.style={-{Stealth[length=5pt]},thick,color=accentblue}]
    \node[box,fill=midgray!60]         (x) {Clean\\image $x$};
    \node[box,fill=pgdred!20,right=0.9cm of x]    (g) {$\nabla$loss\\(surrogate)};
    \node[box,fill=snntemporal!20,right=0.9cm of g] (p) {Perturb\\$x{+}\delta$};
    \node[box,fill=snnrate!20,right=0.9cm of p]   (e) {Encode\\$\to$Spikes};
    \draw[arr](x)--(g);\draw[arr](g)--(p);\draw[arr](p)--(e);
  \end{tikzpicture}
  \end{center}
  {\footnotesize\itshape Straight-through estimator for temporal encoding
  (non-differentiable round/argmax bypassed at attack time).}

  \vspace{0.35em}
  \textbf{\color{pgdred}FGSM}\enspace
  {\small $x_\text{adv}=x+\varepsilon\!\cdot\!\operatorname{sign}(\nabla_x\mathcal{L})$.
  Single-step, worst-case in $\ell_\infty$ ball.}

  \vspace{0.2em}
  \textbf{\color{pgdred}PGD}\enspace
  {\small 10-step iterative with random init.
  Strictly stronger than FGSM.}

  \vspace{0.3em}
  \textbf{\color{accentblue}Surrogate Gradient:}
  {\small $\displaystyle\sigma'(U)=\frac{\text{slope}}{(1+|\text{slope}\cdot U|)^{2}}$}
\end{block}

\begin{block}{Key Findings}
  \colorbox{snntemporal!18}{\parbox{\dimexpr\linewidth-8pt\relax}{%
    \small\textbf{\color{snntemporal}1.}
    \textbf{Temporal coding is consistently more robust} than rate coding
    under FGSM and PGD across both datasets.}}

  \vspace{0.28em}
  \colorbox{snnrate!18}{\parbox{\dimexpr\linewidth-8pt\relax}{%
    \small\textbf{\color{snnrate}2.}
    \textbf{SNNs outperform ANNs/CNNs} under adversarial perturbation
    despite lower clean accuracy --- spike bottleneck acts as implicit
    regularisation.}}

  \vspace{0.28em}
  \colorbox{pgdred!10}{\parbox{\dimexpr\linewidth-8pt\relax}{%
    \small\textbf{\color{pgdred}3.}
    \textbf{PGD is strictly stronger.} Accuracy gap between encoding
    strategies widens under PGD, amplifying the temporal advantage.}}

  \vspace{0.28em}
  \colorbox{cnnpurple!12}{\parbox{\dimexpr\linewidth-8pt\relax}{%
    \small\textbf{\color{cnnpurple}4.}
    \textbf{CIFAR-10 is harder.} All models degrade faster but relative
    ordering (Temporal $>$ Rate $>$ CNN) holds.}}
\end{block}

\begin{block}{Conclusion}
  \colorbox{headerblue}{\parbox{\dimexpr\linewidth-8pt\relax}{%
    \small\color{white}
    Spike encoding \textbf{fundamentally determines adversarial robustness}.
    Temporal (latency) coding distributes information across spike
    \emph{timing} rather than rate, making perturbations harder to exploit.
    SNNs broadly resist adversarial inputs better than ANNs, suggesting
    neuromorphic architectures carry inherent security advantages for
    safety-critical deployments.}}
